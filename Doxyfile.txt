# Die Größe des Symbol-Lookup-Caches kann mit LOOKUP_CACHE_SIZE festgelegt
# werden. Dieser Cache wird verwendet, um Symbole anhand ihres Namens und
# Bereichs aufzulösen. Da dies ein aufwendiger Prozess sein kann und dasselbe
# Symbol oft mehrmals im Code erscheint, halt Doxygen einen Cache mit vorab
# aufgelösten Symbolen. Wenn der Cache zu klein ist, wird Doxygen langsamer.
# Wenn der Cache zu groß ist, wird Speicher verschwendet. Die Cache-Größe wird
# durch diese Formel gegeben: 2^(16+LOOKUP_CACHE_SIZE). Der gültige Bereich ist
# 0..9, der Standard ist 0, was einer Cache-Größe von 2^16=65536 Symbolen
# entspricht. Am Ende eines Laufs meldet Doxygen die Cache-Nutzung und schlagt
# die optimale Cache-Größe aus Geschwindigkeitssicht vor.
# Mindestwert: 0, Höchstwert: 9, Standardwert: 0.

LOOKUP_CACHE_SIZE      = 0

# parallele Verarbeitung effektiv deaktiviert. Bitte melden Sie alle Probleme,
